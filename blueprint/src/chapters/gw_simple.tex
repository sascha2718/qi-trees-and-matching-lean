\chapter{The simple classification}
\label{ch:gw-simple}

We now assemble the simple-support classification. In the chain regime, the quantised geometric
law has small potential, so \cref{ch:iid} matches the labels and the transfer theorem of
\cref{ch:chain-simple} produces a quasi-isometry. The bushy regime uses the same argument with
shape labels. We finish by establishing the geometric obstructions that separate the resulting
classes.

\section{The potential of the quantised law}

\begin{lemma}[Chain potential bound]
\label{thm:eta-bound}
\lean{ChainClasses.quantised_eta_le}
\leanok
\uses{thm:quantised-law, def:etaG}
For every $\theta_1\in(0,1)$ there is $D_0=D_0(\theta_1)\geq5$ such that for every integer
$D\geq D_0$ the law $p^{(D)}$ of \cref{thm:quantised-law} satisfies
\[
 \eta_{\mathsf P,5/2}\bigl(p^{(D)}\bigr)\leq16\,\theta_1^{D(D-5/2)}\leq10^{-4}.
\]
\end{lemma}

\begin{proof}
\leanok
\uses{thm:quantised-law, thm:dominance}
The class indices form the path graph $\mathsf P$. By \cref{thm:quantised-law}, the zero class
has mass $1-\theta_1^{D-1}$, while the positive classes satisfy the doubly exponential bound
$p_j^{(D)}\leq\theta_1^{D^j-1}$. We apply \cref{thm:dominance} with the zero class as the
distinguished vertex. Its non-neighbour mass and the remaining summands are both dominated by
geometric series in the displayed tail bounds. For sufficiently large $D$, this gives
$16\theta_1^{D(D-5/2)}$, which is at most $10^{-4}$ after increasing $D_0(\theta_1)$.
\end{proof}

\begin{proof}[Proof of \cref{thm:twovalue}]
\leanok
\proves{thm:twovalue}
\uses{thm:eta-bound, thm:chain-classes, thm:transfer, thm:matching, thm:level, thm:isometry}
Quantise both label fields at scale $D$ by \cref{thm:chain-classes}. Their common law has
potential below $10^{-4}$ by \cref{thm:eta-bound}, so \cref{thm:matching} supplies, with
probability at least $1-16\eta$, an automorphism of $\bbB$ matching the two quantised fields at
every vertex. Matched labels lie in equal or adjacent classes, hence are comparable within
$D^2$ by \cref{thm:level}. The transfer lemma \cref{thm:transfer} converts the relabelled field
into a $(D^2+3)$-quasi-isometry, and \cref{thm:isometry} identifies the relabelled tree with
the original. Substituting the estimate of \cref{thm:eta-bound} gives the stated failure rate.
The events that such a quasi-isometry exists increase when the permitted constant increases, and
their probabilities tend to one as $D\to\infty$. Their union therefore has probability one.
\end{proof}

\begin{proposition}[Universality in the chain regime]
\label{thm:cross-law}
\lean{ChainClasses.crosslaw_ae_tree, ChainClasses.exists_crosslaw_rate_tree}
\leanok
\uses{thm:chain-coupling, thm:transfer, thm:matching}
Let $\theta,\theta'$ satisfy $\theta_1+\theta_2=1$ and $\theta_1'+\theta_2'=1$ with
$\theta_1,\theta_1'\in(0,1)$, and let $\cT(\omega),\cT'(\omega')$ be independent Galton--Watson
trees with these laws. Then almost surely $\cT(\omega)\simeq\cT'(\omega')$, with a rate at the
constant $\gamma_*D^2+3$, where $\gamma$ is as in \cref{thm:chain-coupling}.
\end{proposition}

\begin{proof}
\leanok
\uses{thm:chain-coupling, thm:eta-bound, thm:transfer}
Use \cref{thm:chain-coupling} to relabel the second geometric field so that its class labels have
the same law as the quantised labels of the first field. The matching theorem can therefore be
applied to these two identically distributed label fields. Whenever two labels match, their
underlying chain lengths are comparable by the ratio bound in the coupling. After relabelling the
branch points by the matching automorphism, \cref{thm:transfer} applies at the real constant
$\gamma_*D^2$. The failure probability tends to zero with $D$, which yields the almost sure
conclusion.
\end{proof}

\section{The bushy regime}

\begin{lemma}[Shrinking a shape into a support]
\label{thm:shape-shrink}
\lean{ChainClasses.markedQI_shape_shrink}
\leanok
\uses{def:shape}
Let $\theta'$ be supported on $\set{0,1,2}$ and supercritical with $\theta_0'>0$, with shape
law $\mu'$. Every shape $\sigma$ admits a support fix $\sigma^\circ\in\supp\mu'$ with
$\abs{\sigma^\circ}\leq2\abs\sigma$ and a $2$-marked quasi-isometry to $\sigma$, and a shape
may be shrunk into the support at a controlled scale.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:shape-iid}
The support fix uses only offspring values $0$ and $2$, both of which have positive mass in the
bushy regime. Attach a one-vertex bush at every bare interior neck vertex. Inside each existing
bush, give every unary vertex a second child that is a leaf. The resulting shape
$\sigma^\circ$ belongs to $\supp\mu'$ and has at most twice as many vertices as $\sigma$.
The natural address map changes only the choice of child at a fixed vertex, so it defines the
claimed $2$-marked quasi-isometry.

For the scale-dependent statement, first contract the realised shape into parts of the chosen
size. Binarise the contracted tree, turn its marked root-to-exit path back into a neck shape, and
apply the preceding support fix. The marked quasi-isometries for these four operations compose to
give the asserted shrinking map and size bound.
\end{proof}

\begin{lemma}[Shape potential bound]
\label{thm:shape-eta}
\lean{ChainClasses.shape_eta_final}
\leanok
\uses{def:shape-net, thm:shape-mass, thm:dilution}
There are $c=c(\theta)>0$ and $D_0=D_0(\theta)$ such that for every integer $D\geq D_0$,
\[
 \eta_{G_D,5/2}(\mu_D)\leq e^{-cD^{2}}\leq10^{-4}.
\]
Moreover the class of the one-vertex shape carries $\mu_D$-mass at least
$1-e^{-cD^{2}}\geq\tfrac12$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:shape-mass, thm:dilution, thm:shape-net, thm:dominance}
Every shape of size at most $D^2$ admits a $D$-marked collapse to the one-vertex shape. The
exponential tail in \cref{thm:shape-mass} therefore shows that the corresponding class has mass
at least $1-e^{-cD^2}$. For the remaining classes, group representatives by size. The entropy
bound of \cref{thm:dilution} controls the number of representatives in each group, while the
pointwise lower mass bound of \cref{thm:shape-mass} controls the denominator in each potential
summand. Summing the resulting exponential series and applying \cref{thm:dominance} gives
$\eta_{G_D,5/2}(\mu_D)\leq e^{-cD^2}$. Increasing $D_0$ gives the numerical threshold.
\end{proof}

\begin{lemma}[Shape coupling]
\label{thm:shape-coupling}
\lean{ChainClasses.exists_isShapeCoupling_etaS}
\leanok
Let $\theta,\theta'$ be supercritical, supported on $\set{0,1,2}$, with $\theta_0,\theta_0'>0$
and shape laws $\mu,\mu'$. There is $D_1=D_1(\theta,\theta')$ and, for $D\geq D_1$, a coupling
of the two shape laws over a common label graph whose potential is again below $10^{-4}$, whose
matched pairs admit a marked quasi-isometry at a controlled scale.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:shape-eta, thm:shape-shrink}
First use \cref{thm:shape-shrink} to place both shape laws on a common family of supported
representatives. Order the remaining shapes by size and match their masses greedily from large
to small. Any surplus is passed to the next smaller representative. The exponential tail ensures
that this descending cascade exhausts both marginals and produces a coupling $q$ of $\mu$ and
$\mu'$.

The shrinking and net maps make every paired shape comparable at the required marked scale. We
take the supported pairs of the coupling as vertices of the common label graph. Its incompatible
mass is controlled by the same tail and capacity estimates used in \cref{thm:shape-eta}. Entropy
dilution then bounds the number of possible tail pairs, and the resulting potential is below
$10^{-4}$ for all sufficiently large $D$.
\end{proof}

\begin{proposition}[Universality in the bushy regime]
\label{thm:hairy}
\lean{ChainClasses.hairy_ae_tree, ChainClasses.hairy_rate_tree_scale}
\leanok
\uses{thm:shape-coupling, thm:glued-transfer, thm:matching}
Let $\theta,\theta'$ be supercritical, supported on $\set{0,1,2}$, with $\theta_0,\theta_0'>0$,
and let $\cT,\cT'$ be independent Galton--Watson trees with these laws conditioned on infinite
diameter. Then almost surely $\cT\simeq\cT'$, with an explicit rate.
\end{proposition}

\begin{proof}
\leanok
\uses{thm:shape-iid, thm:shape-eta, thm:shape-coupling, thm:matching, thm:glued-transfer}
By \cref{thm:shape-iid}, both trees are assemblies of independent shape fields. For one law,
\cref{thm:shape-eta} puts the net labels below the matching threshold. For two laws,
\cref{thm:shape-coupling} supplies a common label graph with the same property. Thus
\cref{thm:matching} gives, with probability tending to one as $D\to\infty$, an automorphism of
$\bbB$ that matches every pair of shape labels. The definition of the graph converts each
matched label pair into a marked quasi-isometry between the corresponding shapes. After applying
the automorphism to one assembly, \cref{thm:glued-transfer} combines these maps into a
quasi-isometry of the complete trees. Taking the union over $D$ gives the almost sure statement.
\end{proof}

\section{The obstructions}

\begin{lemma}[Stability in a tree]
\label{thm:tree-stability}
\lean{BranchingProcess.exists_between_dist_le}
\leanok
\uses{def:qi}
Let $T,T'$ be trees, let $\psi\colon T\to T'$ be a $D$-quasi-isometry, and let $x,y\in T$.
Every vertex of the geodesic $[\psi(x),\psi(y)]$ lies within distance $D$ of the image under
$\psi$ of the geodesic $[x,y]$.
\end{lemma}

\begin{proof}
\leanok
List the consecutive vertices $x=x_0,\dots,x_n=y$ of $[x,y]$. Their images form a discrete
path from $\psi(x)$ to $\psi(y)$ in which consecutive terms are at distance at most $2D$.
Fix $m\in[\psi(x),\psi(y)]$. Removing $m$ separates the two endpoints, so either one image is
already within distance $D$ of $m$, or two consecutive images lie in different components of
$T'\setminus\set{m}$. In the latter case their geodesic passes through $m$ and has length at most
$2D$, which again places $m$ within distance $D$ of one of them. That image is the image of a
vertex on $[x,y]$.
\end{proof}

\begin{proposition}[Hairs against lines]
\label{thm:hair-separation}
\lean{BranchingProcess.not_quasiIsometric_of_hair_of_line}
\leanok
Let $T,T'$ be trees. If $T$ has hairs of unbounded depth and every vertex of $T'$ lies within
distance $R$ of a line of $T'$, then $T\not\simeq T'$.
\end{proposition}

\begin{proof}
\leanok
\uses{thm:tree-stability}
Suppose that $\psi\colon T\to T'$ is a $D$-quasi-isometry. Choose a hair $P$ of depth
$h>D(D+R)+D^2$, attached at $c$, and let $x$ be its endpoint. By hypothesis, $\psi(x)$ lies
within distance $R$ of a line $L\subseteq T'$. Move along $L$ from a nearest point to
$\psi(x)$ in the direction away from $\psi(c)$, and use coarse density to choose preimages in
$T$. The lower quasi-isometry inequality forces such preimages to leave the finite hair through
$c$. On the other hand, \cref{thm:tree-stability} keeps the relevant segment within distance
$D$ of $\psi([c,x])$. Combining the two distance bounds places an off-hair preimage closer to
$x$ than passage through $c$ permits, contradicting the choice of $h$.
\end{proof}

\begin{proposition}[Three rays against the ray]
\label{thm:three-rays}
\lean{BranchingProcess.not_quasiIsometric_rayGraph}
\leanok
Let $T$ be a tree containing three rays that pairwise meet only in their common initial vertex.
Then $T$ is not quasi-isometric to the ray $\bbN_0$.
\end{proposition}

\begin{proof}
\leanok
\uses{thm:tree-stability}
Suppose that $\psi\colon T\to\bbN_0$ is a $D$-quasi-isometry. Let $v$ be the common initial
vertex and choose $a_i$ on the three rays at a common distance $r>2D^2$ from $v$. The four
points $\psi(v),\psi(a_1),\psi(a_2),\psi(a_3)$ are linearly ordered, and the lower distance
bound makes the last three distinct from $\psi(v)$. Two lie on the same side of $\psi(v)$, so
for some $i\neq j$, the nearer point $\psi(a_i)$ lies between $\psi(v)$ and $\psi(a_j)$. By
\cref{thm:tree-stability}, it lies within distance $D$ of the image of a vertex $x$ on
$[v,a_j]$. This geodesic is contained in the $j$th ray, so the disjointness of the rays gives
$d(a_i,x)\geq r$. The lower quasi-isometry inequality, however, gives
$d(a_i,x)\leq D^2+D^2=2D^2$, a contradiction.
\end{proof}

\begin{lemma}[The regimes against the obstructions]
\label{thm:regime-obstructions}
\lean{ChainClasses.unbounded_hairs_ae, ChainClasses.nearLine_inTree}
\leanok
Let $\theta$ be supercritical and supported on $\set{0,1,2}$, and let $\cT$ be a Galton--Watson
tree with law $\theta$ conditioned on infinite diameter. Almost surely $\cT$ carries three rays
meeting only at their common initial vertex. In the chain regime every vertex lies within
bounded distance of a line, while in the bushy regime $\cT$ has hairs of unbounded depth.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:hair-separation, thm:three-rays}
Supercriticality forces $\theta_2>0$. Write $\cT^*$ for $\cT$ when $\theta_0=0$ and for
its skeleton otherwise. In both cases it is a Galton--Watson tree with offspring in
$\set{1,2}$ and with positive probability of two children. Along every infinite line of descent,
the branching property implies that split vertices occur infinitely often. Choosing a split below
a split supplies three rays with only their initial vertex in common.

In the chain regime the tree has no leaves. Below the first split, every vertex lies on a line:
one half follows an infinite descendant ray, while the other climbs to an ancestral split and
leaves through a different infinite child. The finitely many vertices above the first split are
within its depth of such a line. In the bushy regime, the Harris decomposition attaches
independent subcritical bushes along infinitely many skeleton vertices. Every finite hair pattern
has positive probability, so independence and recurrence imply that hairs of arbitrarily large
depth occur almost surely.
\end{proof}

\begin{proposition}[Bottleneck obstruction]
\label{thm:bottleneck}
\lean{ChainClasses.not_isQIWith_binary_of_thinBalls}
\leanok
Let $\cT$ be a tree such that for every $\ell\in\bbN$ there is a vertex $v_\ell$ whose ball
$B_{\cT}(v_\ell,\ell)$ is isometric to a segment of $\bbZ$. Then $\cT$ is not quasi-isometric
to $\bbB$.
\end{proposition}

\begin{proof}
\leanok
\uses{thm:tree-stability}
Suppose that $\psi\colon\cT\to\bbB$ is a $D$-quasi-isometry. Coarse density and the lower
distance inequality imply that a ball of radius $r$ around $\psi(v)$ is covered by the
$D$-neighbourhood of the image of a ball around $v$ whose radius is linear in $r$. Choose
$v=v_\ell$ so that this source ball is contained in the segment
$B_{\cT}(v_\ell,\ell)$. It then has only linearly many vertices, and each of their
$D$-neighbourhoods in $\bbB$ has uniformly bounded size. Hence the target ball has at most
linear growth in $r$. For sufficiently large $\ell$ this contradicts the exponential growth of
balls in $\bbB$.
\end{proof}

\begin{theorem}[Separation from the binary tree and the ray]
\label{thm:converse}
\lean{ChainClasses.converse_ae}
\leanok
\uses{thm:bottleneck, thm:three-rays, thm:regime-obstructions}
Let $\theta$ satisfy $\theta_1+\theta_2=1$ with $0<\theta_1<1$, and let $\cT$ be a
Galton--Watson tree with law $\theta$. Then almost surely $\cT$ is quasi-isometric neither to
the binary tree nor to the ray $\bbN_0$.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:bottleneck, thm:three-rays}
The geometric chain lengths have unbounded support and occur independently over the branch
points. Consequently, for every $\ell$, some chain contains a subsegment of length $2\ell$ whose
central ball is isometric to a segment of $\bbZ$. The bottleneck obstruction
\cref{thm:bottleneck} therefore excludes the binary tree. The same realisation has split vertices
at arbitrarily large depths, and below a split below a split it contains three rays meeting only
at their common initial vertex. The obstruction \cref{thm:three-rays} excludes the ray.
\end{proof}
