\chapter{The matching engine at general arity}
\label{ch:engine}

This chapter proves the matching theorem for tree-indexed Markov label fields. The one-law
argument replaces the divergent cross-type recursion by a finite family of normalised screens.
A grammar records how these screens evolve down the binary tree, and a renewal argument shows
that the live part of the grammar is nilpotent. We then extend the construction to two counter
laws with a common core by adjoining finitely many marker states. The complete development is
formalised in \texttt{GraphMarkovMatching}, which is self-contained over its vendored support
layer.

\section{The matching theorems}

\begin{theorem}[Two-law matching]
\label{thm:composite-matching}
\lean{GraphMarkovMatching.Composite.composite_failure_bound_massfree}
\leanok
\uses{def:etaG}
Let two counter laws share a common core. Under the potential hypothesis on the label graph,
the probability that no automorphism matches the two tree-indexed label fields is bounded by an
explicit multiple of the potential, uniformly in the height, and the same bound holds along the
infinite tree.
\end{theorem}

\begin{corollary}[One-law matching]
\label{thm:main-matching}
\lean{GraphMarkovMatching.varying_failure_le}
\leanok
Specialising both sides to the same counter law gives the one-law matching theorem, with the
failure probability bounded by $K_c\eta$ for an explicit constant.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:composite-matching, thm:numeric}
Apply \cref{thm:composite-matching} with the same counter law on both sides. There are then no
exceptional entrances to encode, so the tagged grammar reduces to the ordinary grammar and no
marker letter is reachable. The common-core constants specialise to the one-law constants of
\cref{thm:numeric}, and the general failure estimate becomes $K_c\eta$.
\end{proof}

\begin{corollary}[Exponential three-label example]
\label{thm:exponential}
\lean{GraphMarkovMatching.varying_exp3_failure_le}
\leanok
Fix $N\geq2$ and a law $\nu$ supported on a nonempty subset of $\set{2,\dots,N}$, let $G$ be the
path $0-1-2$, and for real $D\geq5$ let $\mu_D$ have masses $p_1=e^{-D}$, $p_2=e^{-D^2}$ and
$p_0=1-p_1-p_2$. Then the matching hypothesis holds for all large $D$.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:main-matching}
The compatible masses on the path are
\[
 b(0)=1-p_2,\qquad b(1)=1,\qquad b(2)=p_1+p_2.
\]
Thus the middle state contributes nothing to the potential. The state $0$ has incompatible mass
$p_2=e^{-D^2}$ and a denominator bounded away from zero. At state $2$, the small denominator
is compensated by its still smaller mass $p_2$. For $D\geq5$, both contributions are bounded
by a fixed multiple of $e^{-D}$. The tilt terms are bounded for the same reason, so the
hypotheses of \cref{thm:main-matching} hold once $D$ is sufficiently large.
\end{proof}

\begin{corollary}[Double-exponential law on the half-line]
\label{thm:doubleexp}
\lean{GraphMarkovMatching.varying_doubleExp_failure_le}
\leanok
The same holds for the path on the nonnegative integers with $p_k=e^{-D^k}$ for $k\geq1$ and
$p_0=1-\sum_{k\geq1}p_k$.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:main-matching, thm:double-exp}
For $k\geq1$,
\[
 \frac{p_{k+1}}{p_k}=e^{-D^k(D-1)}\leq e^{-20}.
\]
The positive-index masses and all their tails are therefore bounded by geometric series. In
particular, $\sum_{k\geq1}p_k<1/10$ and $p_0\geq1/2$. Apply the local-dominance estimate on
the half-line with $0$ as the distinguished vertex. The term at $0$ is controlled by the tail
beyond its neighbour, and every other term by the adjacent masses. Summing these bounds gives a
fixed multiple of $e^{-D(D-5/2)}$, which tends to zero and hence satisfies the one-law matching
hypothesis for large $D$.
\end{proof}

\section{Potential-theoretic estimates}

\begin{lemma}[Zero interface]
\label{thm:zero-interface}
\lean{GraphMarkovMatching.qE_zero_le}
\leanok
\uses{def:etaG}
Suppose $\mu(0)\geq1/2$ and write $\eta:=\eta_{G,\alpha}(\mu)$. Then
$\mu(V\setminus B_G(0,1))\leq2\eta$, and the far tilt and tilt moment sums are bounded by
explicit multiples of $\eta$.
\end{lemma}

\begin{proof}
\leanok
We may assume $\eta<\infty$, since otherwise the conclusions are immediate. Hence every state
of positive mass has positive compatible mass $b(v)$. If $v\notin B_G(0,1)$, then $0$ is
incompatible with $v$, so
$1-b(v)\geq\mu(0)\geq1/2$. The summand of $v$ in $\eta$ is therefore at least
$\mu(v)/2$. Summing over the far states gives
$\mu(V\setminus B_G(0,1))\leq2\eta$. The far-tilt and moment estimates follow by comparing
their summands pointwise with the same potential summands and then summing.
\end{proof}

\begin{theorem}[Four-law inequality]
\label{thm:four-law}
\lean{GraphMarkovMatching.PhiD_fourlaw_le}
\leanok
\uses{def:potential, def:square}
Let $\alpha\geq1$ and $\delta,L,K>0$ satisfy the standing hypothesis, let $R$ be a symmetric
relation on a countable set, and let $\mu_0,\mu_1,\nu_0,\nu_1$ be laws. If $M$ bounds all eight
directed potentials between the two source and two target laws, then the potential of the
symmetrised square of the pair laws is bounded by an explicit function of $M$.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:pointwise-fourlaw}
Fix a source pair $(x_0,x_1)$. In column $j$, let the two directed bad degrees be ordered by
$q_0^j\leq q_1^j$, exchanging the row names in that column if necessary. This gives four
possible combinations of the two column orderings. In each case,
\cref{thm:pointwise-fourlaw} bounds the square weight by four one-sided weights and a quadratic
remainder. Integrate this inequality against $\mu_0\otimes\mu_1$. Each linear term is one of
the eight directed potentials and is therefore at most $M$. The products in the remainder
factor under the product source law and are bounded by $M^2$. Collecting the coefficients gives
the claimed explicit function of $M$.
\end{proof}

\begin{proposition}[Pointwise four-law bound]
\label{thm:pointwise-fourlaw}
\lean{GraphMarkovMatching.phi_Q_split_four}
\leanok
At a fixed pair $(x_0,x_1)$ with all four bad degrees below one, the square bad degree is below
one and the weight of the square is bounded by a half-sum of the four one-sided weights plus a
quadratic remainder.
\end{proposition}

\begin{proof}
\leanok
\uses{def:potential}
For $j=0,1$, set
$d_j=\max(q_0^j,q_1^j)$ and $e_j=\min(q_0^j,q_1^j)$. The straight pairing fails with a degree
controlled by one choice from each column, while the crossed pairing uses the opposite choices.
Taking the union of the two pairing events gives a positive lower bound for the square good
degree and hence shows that its bad degree is below one. Apply the tangent-line bound to the terms
linear in the $d_j,e_j$ and the chord bound to the remaining product. The linear contribution is
the half-sum of the four directed one-sided weights, and every term left over contains two bad
degrees, giving the quadratic remainder.
\end{proof}

\section{The one-law transfer system}

\begin{proposition}[The frozen cross-type potential is infinite]
\label{thm:obstruction}
\lean{GraphMarkovMatching.crossed_context_potential_top}
\leanok
\uses{def:potential}
Let $\nu=\delta_3$ and let $v_f$ be any state with $\mu(v_f)>0$ and $v_f\not\sim0$. Then the
directed potential from the frozen cross type into the fresh law is infinite at every height.
\end{proposition}

\begin{proof}
\leanok
Fix a height $m$. Place the incompatible state $v_f$ at the frozen root and follow the unique
counter pattern forced by the $\delta_3$ cascade below it. Complete the other coordinates by any
supported auxiliary labelling. This source configuration has positive mass. However, every target
configuration in the fresh law has a root state incompatible with $v_f$ along the prescribed
cross type, so its target good degree is zero. The corresponding directed-potential summand has
positive numerator and zero denominator, and is therefore infinite. The construction works at
every height.
\end{proof}

This is why the recursion is carried over screens rather than over the naive cross types.

\begin{lemma}[One-step decomposition]
\label{thm:one-step}
\lean{GraphMarkovMatching.rE_succ_branch}
\leanok
Fix a height-$(h+1)$ source point
\[
 x=\mathrm{branch}((v,k);x^1,x^2)
\]
with root state $v$ and child pair $xp$. The good and bad degrees at $x$ decompose over the root
state and the two children.
\end{lemma}

\begin{proof}
\leanok
Write the target law at height $h+1$ as the pushforward of its root-state law and its
height-$h$ child-pair law under the branch map. A target branch is compatible with $x$ precisely
when its root state is compatible with $v$ and its child pair is compatible with $xp$ under one
of the permitted pairings. The root and child data are independent under the kernel. Computing
the good degree through the pushforward therefore factors it into the root-state contribution and
the two child contributions. Subtracting from one gives the corresponding bad-degree
decomposition.
\end{proof}

\begin{lemma}[Screening a fresh mixture]
\label{thm:fresh-mixture}
\lean{GraphMarkovMatching.rE_bind}
\leanok
Let $\rho=\sum_a w_a\rho_a$ be a countable mixture of laws. Then $r_\rho(z)=\sum_a w_ar_{\rho_a}(z)$
for every point $z$, so $r_\rho(z)=0$ if and only if $r_{\rho_a}(z)=0$ for every component with
$w_a>0$.
\end{lemma}

\begin{proof}
\leanok
For fixed $z$, the good degree is the target mass of the set of points compatible with $z$.
Measure is linear in a countable mixture, and Fubini's theorem for the resulting nonnegative sums
gives
$r_\rho(z)=\sum_a w_a r_{\rho_a}(z)$. Every summand is nonnegative. Hence the sum vanishes
precisely when $r_{\rho_a}(z)=0$ for every index with $w_a>0$.
\end{proof}

\begin{definition}[Restricted quantities]
\label{def:restricted}
\lean{GraphMarkovMatching.PhiDres}
\leanok
\uses{def:potential}
For laws $\rho_s,\rho_t$ and a relation $R$, the restricted potential sums the weight only over
points at which the target good degree is positive. The restricted weight and the zero mass are
defined on the same restriction.
\end{definition}

\begin{definition}[Normalised screens]
\label{def:screen}
\lean{GraphMarkovMatching.screenE}
\leanok
\uses{def:restricted}
Let $\rho_s$ be a cell law, $\boldsymbol z$ a finite list of target laws, and
$g\colon\cX\to[0,\infty]$ a normalisation. The screen $\cE(\rho_s;\boldsymbol z\!\not\to g)$
integrates $g$ against $\rho_s$ over the points that are dead against every member of
$\boldsymbol z$.
\end{definition}

\begin{lemma}[Diagonal pruning]
\label{thm:pruning}
\lean{GraphMarkovMatching.screenE_self_mem}
\leanok
\uses{def:screen}
If the cell law is itself a member of the zero list and $R$ is reflexive, then the screen
vanishes for every normalisation.
\end{lemma}

\begin{proof}
\leanok
Fix $x$ in the support of $\rho_s$. Reflexivity makes $x$ compatible with itself, so
$r_{\rho_s}(x)\geq\rho_s(x)>0$. Since $\rho_s$ occurs in the zero list, $x$ is not dead
against every member of that list. The dead-event indicator in the definition of the screen is
therefore zero at every point of positive source mass. Its integral vanishes.
\end{proof}

\begin{lemma}[Tilt pruning]
\label{thm:tilt-pruning}
\lean{GraphMarkovMatching.screenE_tilt_self}
\leanok
\uses{def:screen}
If the normalisation is the restricted tilt of a member of the zero list, the screen vanishes.
\end{lemma}

\begin{proof}
\leanok
Fix $x$. If $r_\rho(x)=0$, the restricted tilt used as the normalisation is zero at $x$. If
$r_\rho(x)>0$, then $x$ is alive against the member $\rho$ of the zero list, so the dead-event
indicator is zero. In either case the integrand vanishes. Thus the complete screen is zero.
\end{proof}

\begin{lemma}[Screened $2\times2$ classification]
\label{thm:hall}
\lean{GraphMarkovMatching.rE_square_eq_zero_iff_hall}
\leanok
\uses{def:screen}
The square good degree vanishes exactly when one source child is dead against both target laws,
or the Hall condition fails on the $2\times2$ array. The surviving cases split into an explicit
sum of resolved terms.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:one-step}
Arrange the four one-sided child degrees in a $2\times2$ array, with source children indexing
rows and target laws indexing columns. The square compatibility event is the union of the straight
and crossed pairing events. Its degree vanishes if and only if both diagonals have zero product.
This happens precisely when some source row is zero against both targets, or when the positive
entries fail Hall's condition. On the complement, partition according to the first positive
diagonal and the remaining zero entries. These disjoint cases give the explicit resolved terms in
the formula.
\end{proof}

\begin{lemma}[Mixture bound for restricted weights]
\label{thm:mixture-tilt}
\lean{GraphMarkovMatching.WresD_XiBar_le_sum}
\leanok
\uses{def:restricted}
For a pair of height-$h$ full labellings, the restricted weight of the mixed pair law is bounded
by the tilt-weighted sum of the restricted weights of its components.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:fresh-mixture}
Fix a pair point. If its good degree against the mixed target is zero, the restricted weight on
the left is zero. Otherwise, \cref{thm:fresh-mixture} writes this degree as
$\sum_a w_a r_{\rho_a}$. Divide the component contributions by the positive mixed degree. The
resulting coefficients are the component tilts, and the pointwise restricted weight is bounded by
their weighted sum. Integrating over the source pair gives the claim.
\end{proof}

\section{The grammar and the accessible index}

The transfer estimates generate several kinds of target laws and screens. We encode their
possible transitions by a finite grammar. Restricting to objects reachable from the fresh pair
keeps the later recursion finite and removes component letters that cannot occur in a live row.

\begin{definition}[Targets, cascade pairs, successors]
\label{def:grammar}
\lean{GraphMarkovMatching.Tgt}
\leanok
The target alphabet $\cA$ consists of the forced letters $\Zf_j$, the fresh letter $\Ff$, and
the component letters $\Ff_k$. The cascade pair of a counter and the successor relation on
letters are defined by the kernel recursion.
\end{definition}

\begin{lemma}[The bounded alphabet and screen family]
\label{thm:universes}
\lean{GraphMarkovMatching.screenUniv_card_le}
\leanok
\uses{def:grammar}
Fix $N\geq2$ with $S_\nu\subseteq\set{0,\dots,N}$. Then successors stay inside $\cA_N$, full
successors of members of $\cU_N$ stay in $\cU_N$, and
$\abs{\cA_N}\leq2N+3$, $\abs{\cU_N}\leq(2N+3)2^{2N+3}(2N+4)$.
\end{lemma}

\begin{proof}
\leanok
For $j\leq N$, every entry of its cascade pair is either $\Ff$ or a forced letter $\Zf_i$ with
\[
 i\leq\max\set{\floor{j/2},j-\floor{j/2}}\leq N.
\]
Thus a successor of a letter in $\cA_N$ remains in $\cA_N$, and the same holds coordinatewise
for full successors of screens. There are at most $2N+3$ forced, fresh, and component letters.
A formal screen is determined by its cell letter, a subset of the alphabet as its zero list, and
one letter or the trivial choice as its normalisation. Multiplying these choices gives the stated
bound for $\abs{\cU_N}$.
\end{proof}

\begin{definition}[Interpretation of the grammar]
\label{def:interp}
\lean{GraphMarkovMatching.interpT, GraphMarkovMatching.interpScreen}
\leanok
\uses{def:grammar}
At height $h$ the letters are interpreted as the process laws $Z_{j,h}$, $T_h$ and $F_{k,h}$,
and a normalisation letter as the corresponding restricted tilt.
\end{definition}

\begin{definition}[Accessible ordered pairs]
\label{def:ordacc}
\lean{GraphMarkovMatching.OrdAcc}
\leanok
\uses{def:grammar}
The set $\mathfrak O_\nu$ of accessible ordered pairs is the smallest subset of $\cA^2$
containing $(\Ff,\Ff)$ and closed under pair successors.
\end{definition}

\begin{lemma}[Invariants of accessible pairs]
\label{thm:ordacc-inv}
\lean{GraphMarkovMatching.OrdAcc.swap}
\leanok
\uses{def:ordacc}
$\mathfrak O_\nu$ is closed under swapping coordinates and under the two diagonal maps, no
component letter occurs in an accessible pair, and $\mathfrak O_\nu\subseteq\cA_N^2$ when
$S_\nu\subseteq\set{0,\dots,N}$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:universes}
We induct on the derivation of accessibility. The initial pair $(\Ff,\Ff)$ is fixed by the swap
and both diagonal maps. If a pair has any of these symmetries, applying a pair-successor operation
before or after the symmetry gives the same successor pair, so the properties pass through the
closure. The successor rules for $\Ff$ and the forced letters produce only fresh or forced
letters. Since the initial pair contains no component letter, none can appear at a later stage.
The bounded-alphabet assertion follows simultaneously from \cref{thm:universes}.
\end{proof}

\begin{definition}[Accessible screens and the live index]
\label{def:scracc}
\lean{GraphMarkovMatching.ScrAcc}
\leanok
\uses{def:screen, def:ordacc}
A formal screen is a seed when its cell, zero list and normalisation all come from the
successors of one accessible pair. The live index $\mathfrak I_\nu$ consists of the live members
of the closure of the seeds under the successor operations.
\end{definition}

\begin{lemma}[Invariants of accessible screens]
\label{thm:scracc-inv}
\lean{GraphMarkovMatching.ScrAcc.nonempty}
\leanok
\uses{def:scracc}
Every accessible screen has a nonempty zero list, no component letter occurs as its cell, in its
zero list, or as its normalisation, and its cell pairs accessibly with itself and with its
normalisation.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:ordacc-inv}
Again we induct on the construction. A seed is formed from successors of one accessible pair. Its
zero list is nonempty, and \cref{thm:ordacc-inv} ensures that none of its cell, zero-list, or
normalisation letters is a component letter. Each closure operation replaces these letters by
successors or removes zero-list entries, so nonemptiness for a live screen and component-freeness
are preserved. Applying the swap and diagonal invariants of \cref{thm:ordacc-inv} at each step
also shows that the cell is accessible with itself and with the normalisation.
\end{proof}

\begin{definition}[The block matrix]
\label{def:accN}
\lean{GraphMarkovMatching.accN}
\leanok
\uses{def:scracc}
Fix $C_W\in[0,\infty]$. The block matrix on the live index carries $C_W$ from $i$ to each full
successor $j$ of $i$, and zero elsewhere.
\end{definition}

\section{The transfer rows}

We now translate the pointwise decomposition into inequalities for the ordinary potential
coordinates and for the screens. The ordinary rows close up to the current screen supremum. The
screen rows have, in addition, a linear successor term represented by the block matrix.

\begin{lemma}[Restricted four-law]
\label{thm:res-four-law}
\lean{GraphMarkovMatching.PhiDres_fourlaw_le}
\leanok
\uses{def:restricted}
The four-law inequality holds for the restricted potentials, with an added zero-mass term.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:four-law}
Partition the source product into points at which all required target degrees are positive and its
complement. On the first set, the proof of \cref{thm:four-law} applies without change to the
restricted weights. On the complement, the restricted weights vanish but the discarded source
mass must be retained explicitly. This is the added zero-mass term. Integrating the two parts and
combining their bounds gives the restricted four-law inequality.
\end{proof}

\begin{definition}[The cell bound]
\label{def:cellCB}
\lean{GraphMarkovMatching.cellCB}
\leanok
\uses{thm:res-four-law}
For bounds $M,Z,E'$ the cell bound $\mathrm{CB}(M,Z,E')$ collects the linear, quadratic,
zero-mass and screen contributions of one cell.
\end{definition}

\begin{lemma}[One-cell row]
\label{thm:one-cell}
\lean{GraphMarkovMatching.PhiDres_square_le}
\leanok
\uses{def:cellCB, def:screen}
If in addition the four resolved screens are bounded by $E'$, then the restricted potential of
the square is at most the cell bound.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:res-four-law, def:screen}
Write the restricted square potential as the integral of its weight over source pairs alive
against the target product. Apply \cref{thm:res-four-law} to split this integral into the linear
potential terms, the quadratic remainder, and the zero-mass term. The unresolved parts of the
$2\times2$ classification are precisely the four screens in the hypothesis, each at most $E'$.
Substitution of these four bounds produces the expression $\mathrm{CB}(M,Z,E')$.
\end{proof}

\begin{lemma}[Square cell between cascades]
\label{thm:psirow-square}
\lean{GraphMarkovMatching.psiRow_square}
\leanok
\uses{def:interp}
Fix counters $k,j$ and a height $h$. If the ordinary coordinates, zero masses and screens
between the two cascade families are bounded, then so is the square cell between them.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:one-cell}
Interpret the two cascade families at height $h$. Their source and target cells are product laws,
and their symmetrised comparison at height $h+1$ is the square cell in the statement. We apply
\cref{thm:one-cell} to these two product laws. The ordinary-coordinate hypotheses bound the
directed potentials by $M$, the zero hypotheses bound the discarded mass by $Z$, and the four
screen hypotheses give $E'$. The resulting cell bound is exactly the desired estimate.
\end{proof}

\begin{definition}[The one-sided bound]
\label{def:oneSided}
\lean{GraphMarkovMatching.oneSidedBound}
\leanok
For bounds $M,E'$ and a tilt bound $T_\nu$, the one-sided bound is
$\mathrm{OS}(M,E'):=T_\nu\abs{S_\nu}\bigl(4E'(1+\alpha M)+4E'^2\bigr)$.
\end{definition}

\begin{lemma}[One-sided tilt mass]
\label{thm:one-sided}
\lean{GraphMarkovMatching.oneSided_charge_le}
\leanok
\uses{def:oneSided}
Under the row hypotheses, the one-sided contribution at a counter is at most
$\mathrm{OS}(M,E')$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:mixture-tilt}
At a fixed pair point, first bound the event that at least one supported component is dead by the
sum of the finitely many component-dead indicators. Next apply \cref{thm:mixture-tilt} to the
restricted weight of the fresh mixture. Multiplying these two pointwise bounds produces one term
for each supported counter, each controlled by the ordinary and screen bounds $M$ and $E'$.
Summing over at most $\abs{S_\nu}$ counters and inserting the tilt bound $T_\nu$ gives
$\mathrm{OS}(M,E')$.
\end{proof}

\begin{lemma}[The four ordinary rows]
\label{thm:psi-rows}
\lean{GraphMarkovMatching.psiRow_ZZ}
\leanok
\uses{def:cellCB}
With $\Delta:=\mathrm{CB}(M,Z,E')+\mathrm{OS}(M,E')$, each of the four ordinary coordinates at
height $h+1$ is bounded by $\Delta$ under the corresponding hypotheses at height $h$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:psirow-square, thm:one-sided}
Consider one of the four choices of forced or fresh source and target. Its height-$(h+1)$ law is
the pushforward of the corresponding cascade law together with a compatible root state.
\Cref{thm:psirow-square} bounds the child square cell by $\mathrm{CB}(M,Z,E')$. When a fresh
mixture is present, components that are dead at the current point leave a one-sided remainder,
which \cref{thm:one-sided} bounds by $\mathrm{OS}(M,E')$. Adding the two estimates gives
$\Delta$. The same argument applies to all four orientations.
\end{proof}

\begin{lemma}[Descent of a zero list]
\label{thm:descent}
\lean{GraphMarkovMatching.memInd_branch_iff}
\leanok
\uses{def:screen}
For an $\Ff_k$-free zero list and a root state $v\sim0$ with $b(v)>0$, deadness of a branch
point against the list is equivalent to deadness of the child pair against the descended list.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:one-step}
Fix one member of the zero list. By \cref{thm:one-step}, its good degree at the branch point is
the product of the positive root factor $b(v)$ and the appropriate child-pair degree. The branch
point is therefore dead against this member precisely when the child pair is dead against every
member produced by its successor rule. Taking the conjunction over the original zero list gives
deadness against the complete descended list. The absence of component letters ensures that no
additional mixture case occurs.
\end{proof}

\begin{lemma}[Multi-member Hall factorisation]
\label{thm:hall-factor}
\lean{GraphMarkovMatching.hallFactorize}
\leanok
When every component of a member of the product list lies in the scalar list, deadness against
the product list factorises through deadness against the scalar list.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:hall}
At a source pair, joint deadness against the product list can occur in three ways: the first source
coordinate is dead against all scalar components, the second is, or the two coordinates fail
against complementary scalar components. The containment hypothesis ensures that these scalar
events cover every product-list member. Hence the joint-deadness indicator is bounded by the sum
of the two one-sided indicators and the corresponding cross products. Integrating against the
product source law makes the cross products factor, which gives the asserted bound in terms of
scalar deadness.
\end{proof}

\begin{lemma}[The six screen rows]
\label{thm:screen-rows}
\lean{GraphMarkovMatching.interpScreen_forced_step}
\leanok
\uses{def:scracc}
For a formal screen whose cell is forced or fresh, with nonempty $\Ff_k$-free zero list and
$\Ff_k$-free normalisation, the screen at height $h+1$ is bounded by the sum of its successor
screens at height $h$ plus the ledger terms.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:descent, thm:hall-factor}
Suppose first that the cell is forced. Its height-$(h+1)$ law is the branch pushforward of one
cascade law below a fixed compatible root. By \cref{thm:descent}, the zero-list event descends to
the child pair. The Hall factorisation of \cref{thm:hall-factor} then divides this event among the
formal successor screens, whose contributions form the block-matrix term.

For a fresh cell, expand the root law into its counter components and apply the same argument to
each supported component. The components with a compatible root give the successor screens. The
remaining components contribute only the root-tilt and far-mass errors, which are the ledger
terms. The six possible choices of cell and normalisation are covered by these two computations.
\end{proof}

\begin{lemma}[Base bounds]
\label{thm:base}
\lean{GraphMarkovMatching.interpPhi_base}
\leanok
\uses{def:interp}
Suppose $\mu(0)\geq1/2$. At height $0$ every $\Ff_k$-free ordinary coordinate is at most
$\eta+\phi^E_\alpha(q_\mu(0))$, and every admissible formal screen is bounded likewise.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:zero-interface}
At height $0$, a full labelling consists of one state and the matching relation is the original
state compatibility relation. A fresh-to-fresh coordinate is therefore the graph potential
$\eta$. Against a forced target with compatible root, the target degree is one except for the
possible zero-state contribution, whose weight is $\phi^E_\alpha(q_\mu(0))$. This gives the
ordinary bound. Every admissible screen integrates over a subset of the same state space with a
normalisation bounded by the same restricted weight, so the identical two terms also bound the
screen.
\end{proof}

\section{Nilpotence of the accessible screen block}

The screen recursion will close only if successor screens cannot persist indefinitely. The
following renewal language aligns the return times of a source letter with the descendants of the
initial zero list. This alignment forces every sufficiently long accessible path to leave the live
index.

\begin{definition}[Return lengths]
\label{def:renewal}
\lean{GraphMarkovMatching.toFresh}
\leanok
The return lengths of a counter are defined by the recursion mirroring the kernel:
$\mathrm{toF}(k)=\set1$ for $k\leq2$, $\mathrm{toF}(3)=\set{1,2}$, and
$\mathrm{toF}(k)=(\mathrm{toF}(\floor{k/2})+1)\cup(\mathrm{toF}(\ceil{k/2})+1)$ for $k\geq4$.
\end{definition}

\begin{lemma}[Cofinality of the generated monoid]
\label{thm:semigroup}
\lean{GraphMarkovMatching.depths_semigroup}
\leanok
\uses{def:renewal}
There is a threshold $R_0$ such that every multiple of $d_\nu$ at least $R_0$ belongs to
$\Sigma_\nu$.
\end{lemma}

\begin{proof}
\leanok
Divide the finite set of return lengths by their greatest common divisor $d_\nu$. The resulting
generators are coprime. B\'ezout's identity, with its positive and negative parts separated, gives
two consecutive elements of the generated monoid. Every sufficiently large integer is a
nonnegative combination of two consecutive positive integers. Multiplying by $d_\nu$ yields a
threshold beyond which every multiple of $d_\nu$ lies in $\Sigma_\nu$.
\end{proof}

\begin{definition}[Admissible descents]
\label{def:descend}
\lean{GraphMarkovMatching.Descend}
\leanok
\uses{def:grammar}
Write $t\Downarrow_m t'$ when there is a chain of $m$ successor steps from $t$ to $t'$.
\end{definition}

\begin{lemma}[Realisation and decomposition of fresh returns]
\label{thm:descend-realise}
\lean{GraphMarkovMatching.descend_of_mem_toFresh}
\leanok
\uses{def:descend, def:renewal}
The fresh-to-fresh descent lengths are exactly the monoid:
\[
 \set{m\colon\Ff\Downarrow_m\Ff}=\Sigma_\nu.
\]
Every letter admits some descent to $\Ff$, and every return length in $\mathrm{toF}(k)$ with
$k\in S_\nu$ is realised.
\end{lemma}

\begin{proof}
\leanok
\uses{def:renewal}
We first prove realisation by induction on the counter. If the successor set contains the cascade
pair of $k$, then a return through either child counter realises every length prescribed by the
recursive definition of $\mathrm{toF}(k)$. The base counters return in one or two steps exactly as
listed in \cref{def:renewal}. This also shows that every forced letter eventually reaches $\Ff$.

A descent from $\Ff$ first chooses a supported counter and then follows one of its return paths.
Thus every primitive fresh return has length in a generating set of $\Sigma_\nu$. Concatenating
fresh returns gives every finite sum of these lengths. Conversely, split any fresh-to-fresh descent
at its intermediate occurrences of $\Ff$ and apply the realisation result to each segment. Its
length is a sum of generators, proving the equality with $\Sigma_\nu$.
\end{proof}

\begin{lemma}[List coverage]
\label{thm:coverage}
\lean{GraphMarkovMatching.mem_zsucc_iter_of_descend}
\leanok
\uses{def:descend}
If $t\in\boldsymbol z$ and $t\Downarrow_mt'$, then $t'$ lies in the $m$-fold descended list, and
along any chain of full successor screens the zero list after $n$ steps is the $n$-fold descent
of the initial one.
\end{lemma}

\begin{proof}
\leanok
\uses{def:descend}
For $m=0$, the empty descent gives $t'=t\in\boldsymbol z$. For one step, the definition of the
descended list includes every successor of every member. Induction on $m$ advances the membership
through one further list descent at each letter step. Along a chain of full successor screens, the
zero-list coordinate is, by definition, the full descent of the preceding zero list. A second
induction therefore identifies the zero list after $n$ screen steps with the $n$-fold descended
initial list.
\end{proof}

\begin{lemma}[Returns along source chains]
\label{thm:source-ray}
\lean{GraphMarkovMatching.hits_F_of_gpair}
\leanok
Along a chain of successor letters, a cascade entry at counter $k$ returns to $\Ff$ after some
$m\in\mathrm{toF}(k)$, and a fresh letter returns after some renewal depth.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:descend-realise}
We use strong induction on $k$. If the next letter is $\Ff$, the chain returns in one step and
$1\in\mathrm{toF}(k)$. Otherwise it is a forced letter at one of the strictly smaller counters in
the cascade pair of $k$. The inductive hypothesis supplies a return from that counter, and the
recursive definition of $\mathrm{toF}(k)$ adds the initial step to its length. Starting from
$\Ff$, the next letter lies in the cascade pair of some supported counter, so the same argument
produces another fresh letter. Repeating it shows that fresh return times are unbounded along any
successor chain.
\end{proof}

\begin{definition}[Anchored letters and screens]
\label{def:anchored}
\lean{GraphMarkovMatching.Anchored}
\leanok
\uses{def:descend, thm:semigroup}
A letter is anchored at absolute depth $d$ when $t\Downarrow_m\Ff$ implies $d+m\in\Sigma_\nu$. A
screen is anchored at $d$ when its cell and every member of its zero list are.
\end{definition}

\begin{lemma}[Anchoring propagates]
\label{thm:anchor-step}
\lean{GraphMarkovMatching.Anchored.step}
\leanok
\uses{def:anchored}
If $t$ is anchored at $d$ and $t'$ is a successor, then $t'$ is anchored at $d+1$. The same
holds for screens. If $\Ff$ is anchored at $d$, then $d\in\Sigma_\nu$. Every accessible pair
has both coordinates anchored.
\end{lemma}

\begin{proof}
\leanok
\uses{def:anchored}
If $t'$ has a fresh completion of length $m$, prepend the successor step $t\to t'$ to obtain a
fresh completion of $t$ of length $m+1$. Anchoring of $t$ at $d$ gives
$d+1+m\in\Sigma_\nu$, which is precisely anchoring of $t'$ at $d+1$. Apply this argument to
the cell and every zero-list member to obtain the screen statement. For $t=\Ff$, the empty
completion has length zero, so anchoring at $d$ forces $d\in\Sigma_\nu$. Finally, induction over
the accessible-pair and seed constructions anchors all letters at the common depth from which
they are reached.
\end{proof}

\begin{theorem}[No anchored live path]
\label{thm:nilpotence}
\lean{GraphMarkovMatching.no_anchored_live_path}
\leanok
\uses{def:scracc}
There is no infinite chain of formal screens, each a full successor of the last, starting from
an anchored screen with nonempty zero list and never having its cell in its zero list.
Consequently the block matrix on the live index is nilpotent.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:anchor-step, thm:coverage, thm:source-ray, thm:descend-realise, thm:semigroup}
Suppose that $(\sigma_n)_{n\geq0}$ is such a chain, with $\sigma_0$ anchored at depth $d_0$.
By \cref{thm:anchor-step}, $\sigma_n$ is anchored at $d_0+n$. Choose a letter $t$ from the
nonempty zero list of $\sigma_0$ and a descent $t\Downarrow_s\Ff$, which exists by
\cref{thm:descend-realise}. Then \cref{thm:coverage} places $\Ff$ in the zero list of
$\sigma_s$, and anchoring gives $d_0+s\in\Sigma_\nu$.

By \cref{thm:source-ray}, the cell along the screen path equals $\Ff$ at unboundedly many
times. Choose such a time $n\geq s+R_0$, where $R_0$ is the threshold from
\cref{thm:semigroup}. Anchoring of the cell gives $d_0+n\in\Sigma_\nu$. Hence $n-s$ is a
multiple of $d_\nu$ and is at least $R_0$, so $n-s\in\Sigma_\nu$. By
\cref{thm:descend-realise} and \cref{thm:coverage}, the fresh letter already in the zero list at
time $s$ descends to a fresh letter in the zero list at time $n$. The cell and a zero-list member of
$\sigma_n$ are therefore both $\Ff$, contradicting liveness.

A directed cycle in the finite live index could be repeated to form such an infinite path. The
successor graph is thus acyclic. Ordering its vertices by reachability makes the block matrix
strictly triangular, and hence nilpotent.
\end{proof}

\section{The closed recursion}

We now collect the many ordinary coordinates into one scalar and the accessible screens into one
finite vector. The transfer rows yield a monotone two-variable recursion. Nilpotence turns the
linear screen part into a finite invariant sum, after which a joint induction closes both bounds.

\begin{definition}[Ordinary scalar and screen vector]
\label{def:running}
\lean{GraphMarkovMatching.genPsi}
\leanok
\uses{def:ordacc, def:scracc}
Put $\overline\Psi_h:=\sup_{(s,t)\in\mathfrak O_\nu}\Psi_h(s,t)$, let $E_h$ be the vector of
screens over the live index, and $\overline E_h$ its supremum.
\end{definition}

\begin{definition}[Ordinary and screen step functions]
\label{def:step-functions}
\lean{GraphMarkovMatching.genF, GraphMarkovMatching.genG}
\leanok
\uses{def:cellCB, def:oneSided}
Fix a tilt bound and ledger bounds. The ordinary step function $f$ and screen step function $g$
are the explicit increasing functions assembled from the cell and one-sided bounds.
\end{definition}

\begin{lemma}[The ordinary step]
\label{thm:psi-step}
\lean{GraphMarkovMatching.genPsi_step}
\leanok
\uses{def:step-functions}
$\overline\Psi_{h+1}\leq f(\overline\Psi_h,\overline E_h)$ for every $h$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:psi-rows, thm:ordacc-inv}
Fix an accessible pair. By \cref{thm:ordacc-inv}, neither coordinate is a component letter, so
its height-$(h+1)$ comparison is one of the four ordinary rows in \cref{thm:psi-rows}. All
ordinary coordinates at height $h$ are at most $\overline\Psi_h$, and all screens are at most
$\overline E_h$. Monotonicity of the row expression therefore bounds this coordinate by
$f(\overline\Psi_h,\overline E_h)$. Taking the supremum over accessible pairs gives the result.
\end{proof}

\begin{lemma}[The screen step]
\label{thm:e-step}
\lean{GraphMarkovMatching.genE_step}
\leanok
\uses{def:accN, def:step-functions}
For $\mu(0)>0$ and $C_W$ large enough,
$E_{h+1}(i)\leq g(\overline\Psi_h,\overline E_h)+(\Nb E_h)(i)$ for every $h$ and every $i$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:screen-rows, thm:scracc-inv}
Let $i$ be a live accessible screen. The invariants of \cref{thm:scracc-inv} exclude component
letters from its cell, zero list, and normalisation, so it falls under one of the six estimates in
\cref{thm:screen-rows}. Bound every ordinary coordinate and every screen remainder in that
estimate by $\overline\Psi_h$ and $\overline E_h$. The terms independent of the successor screen
are then bounded by $g(\overline\Psi_h,\overline E_h)$. The remaining sum ranges over the full
successors of $i$, each with coefficient $C_W$, and is exactly $(\Nb E_h)(i)$.
\end{proof}

\begin{lemma}[Nilpotent invariant]
\label{thm:invariant}
\lean{GraphMarkovMatching.nilpotent_invariant}
\leanok
Let $\Nb$ be nonnegative on a finite index with $\Nb^{\,r}(u\mathbf1)=0$, and put
$\widehat E:=\sum_{j<r}\Nb^{\,j}(u\mathbf1)$. Then $u\mathbf1+\Nb\widehat E=\widehat E$, and any
vector sequence dominated by the step stays below $\widehat E$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:nilpotence}
Applying $\Nb$ to the finite sum and shifting the index gives
\[
 \Nb\widehat E=\sum_{1\leq j\leq r}\Nb^{\,j}(u\mathbf1)
 =\widehat E-u\mathbf1,
\]
because the final term $\Nb^{\,r}(u\mathbf1)$ vanishes. This proves the displayed invariant
identity. Now suppose $E_h\leq\widehat E$ and the scalar part of the step is at most $u$. By
monotonicity,
$E_{h+1}\leq u\mathbf1+\Nb E_h\leq u\mathbf1+\Nb\widehat E=\widehat E$.
The initial vector is bounded by the first summand $u\mathbf1$, so induction proves invariance.
\end{proof}

\begin{theorem}[The screened recursion closes]
\label{thm:engine}
\lean{GraphMarkovMatching.screened_uniform_bound_mono}
\leanok
Under nilpotence of $\Nb$, monotone step functions, and the size and closure inequalities, the
ordinary scalar stays below $K_c\eta$ and the screen vector below $\widehat E$ at every height.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:invariant, thm:psi-step, thm:e-step}
We prove by induction on $h$ the joint bounds
$\overline\Psi_h\leq K_c\eta$ and $E_h\leq\widehat E$. The base inequalities give the claim at
height zero. Assume it at height $h$. The vector bound implies
$\overline E_h\leq\sup_i\widehat E(i)$. Monotonicity of $f$, followed by the ordinary closure
inequality, gives $\overline\Psi_{h+1}\leq K_c\eta$. Similarly, monotonicity of $g$ bounds the
nonlinear part of the screen step by $u$. The identity in \cref{thm:invariant} then gives
\[
 E_{h+1}\leq u\mathbf1+\Nb E_h
 \leq u\mathbf1+\Nb\widehat E=\widehat E.
\]
This closes the joint induction.
\end{proof}

\begin{lemma}[Geometric bound for the invariant]
\label{thm:geom}
\lean{GraphMarkovMatching.mulVec_le_of_row_sum}
\leanok
If every row sum of $\Nb$ is at most $c$, then $\Nb^{\,j}(u\mathbf1)\leq c^ju\mathbf1$ and the
invariant is at most $(\sum_{j<r}c^j)u\mathbf1$. For the block matrix one may take
$c=C_W\abs{\mathfrak I_\nu}$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:invariant}
If $x\leq b\mathbf1$ entrywise, then for every row $i$,
\[
 (\Nb x)(i)\leq b\sum_j\Nb(i,j)\leq cb.
\]
Thus one matrix application multiplies a constant vector bound by at most $c$. Starting from
$u\mathbf1$ and iterating proves
$\Nb^{\,j}(u\mathbf1)\leq c^ju\mathbf1$. Summing this estimate for $j<r$ gives the bound on
$\widehat E$. For the block matrix, each row has at most $\abs{\mathfrak I_\nu}$ nonzero entries,
all equal to $C_W$, so its row sum is at most $C_W\abs{\mathfrak I_\nu}$.
\end{proof}

\begin{theorem}[Uniform failure bound, closure form]
\label{thm:closure-form}
\lean{GraphMarkovMatching.varying_failure_uniform}
\leanok
With the step functions of \cref{def:step-functions} and $r_\nu=\abs{\mathfrak I_\nu}+1$ from
\cref{thm:nilpotence}, and the ledger bounds in force, the failure probability is bounded
uniformly in the height by the closed form.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:engine, thm:geom, thm:base, thm:nilpotence, def:running, def:accN,
thm:psi-step, thm:e-step}
Use the running scalar and screen vector of \cref{def:running}. Their step inequalities are
\cref{thm:psi-step} and \cref{thm:e-step}, and \cref{thm:base} supplies their initial bounds. The
block matrix is the one from \cref{def:accN}. It is nilpotent by \cref{thm:nilpotence}, and
\cref{thm:geom} bounds the corresponding invariant vector by the stated finite geometric sum.
All hypotheses of \cref{thm:engine} are therefore satisfied, so its uniform scalar bound is the
claimed failure estimate.
\end{proof}

\section{The numerical closure}

\begin{lemma}[Ledger constants]
\label{thm:ledger-constants}
\lean{GraphMarkovMatching.rootTilt_le}
\leanok
If $\sim$ is symmetric and $\mu(0)\geq1/2$, the ledger hypotheses hold with $R_T=8$,
$F_M=2\eta$, $F_T=2\eta$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:zero-interface}
If $v\sim0$, then $0\in B_G(v,1)$ and hence
$b(v)\geq\mu(0)\geq1/2$. Therefore
$b(v)^{-5/2}\leq2^{5/2}\leq8$, which gives the root-tilt constant $R_T$. The mass of states
far from $0$ and their tilt-weighted mass are bounded by the two corresponding conclusions of
\cref{thm:zero-interface}. These give $F_M=F_T=2\eta$.
\end{proof}

\begin{definition}[The explicit constants]
\label{def:constants}
\lean{GraphMarkovMatching.genKcC}
\leanok
\uses{thm:ledger-constants, thm:universes}
The constants $C_W$, $K_c$ and the level bounds are fixed from the tilt bound, the cardinality
bounds $c_N\geq\abs{\mathfrak I_\nu}$ and $n_{\cA}\geq\abs{\cA_N}$, and the support size.
\end{definition}

\begin{theorem}[Numerical closure]
\label{thm:numeric}
\lean{GraphMarkovMatching.varying_failure_le}
\leanok
\uses{def:constants}
If $\sim$ is reflexive and symmetric, $\mu(0)\geq1/2$ and $\eps^{-1}\eta\leq1$, then the size
and closure inequalities hold with the explicit constants and the failure probability is at most
$K_c\eta$.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:closure-form, thm:ledger-constants, def:constants, thm:universes}
Insert the ledger values of \cref{thm:ledger-constants} into the explicit definitions in
\cref{def:constants}. The assumption $\eps^{-1}\eta\leq1$ bounds every higher-order term in the
step functions by its designated multiple of $\eta$. The cardinality estimates in
\cref{thm:universes} bound the row sums and the length of the nilpotent invariant. Substitution
then verifies, in order, the initial-size inequality, the screen closure inequality, and the
ordinary closure inequality required by \cref{thm:closure-form}. Its conclusion is the bound
$K_c\eta$.
\end{proof}

\begin{lemma}[Marginal consistency]
\label{thm:marginal-consistency}
\lean{GraphMarkovMatching.Tlaw_map_restrictLab}
\leanok
Restriction of a height-$(h+1)$ labelling to $\bbB_h$ pushes $T_{h+1}$ forward to $T_h$, so the
finite-height bounds pass to the infinite tree.
\end{lemma}

\begin{proof}
\leanok
Restriction commutes with the branch constructor: removing the last level from a branch is the
branch of the two restricted child labellings. At height zero the assertion is immediate. For the
inductive step, condition on the root counter and state. The Markov kernel makes the two child
labellings conditionally independent with the height-$h$ process law, whose restrictions have the
height-$(h-1)$ law by induction. Reassembling the root mixture gives the required pushforward
identity. Hence the finite-dimensional process laws are consistent.
\end{proof}

\section{Two counter laws with a common core}

The one-law grammar assumes that the same counter can be used on both sides of every comparison.
For two laws, counters in their common support still use the ordinary cascade. An exceptional
counter is instead routed through a finite marker chain to a declared counter on the other side,
after which the process rejoins the common core. We enlarge the grammar by these marker letters
and repeat the nilpotent-recursion argument on the resulting finite index.

\begin{definition}[Composite kernels]
\label{def:composite-kernel}
\lean{GraphMarkovMatching.Composite.compK}
\leanok
\uses{def:grammar}
The label alphabet consists of the plain labels $(v,k)$ together with the marked labels
$(0,\langle p,i\rangle)$ for a declared pair $p$ and a stage $i$ along its designated path. The
composite kernel walks the designated path and rejoins the common core.
\end{definition}

\begin{lemma}[Cluster identity]
\label{thm:cluster}
\lean{GraphMarkovMatching.Composite.mul_repZ_le_cZ}
\leanok
\uses{def:composite-kernel}
For a side $\sigma$, a declared pair $p=(a,b)$ and a height $h$, the marked pair law is the
cascade law at $a$ conditioned on the designated port carrying the label $(0,b)$, an event of
probability $\mu(0)\nu_\sigma(b)>0$.
\end{lemma}

\begin{proof}
\leanok
\uses{def:composite-kernel}
Expand the cascade below a root with counter $a$ using the balanced rule. This determines the
counter at every internal position of the cascade, while the state labels retain their prescribed
conditional laws. The designated port is fresh and independent of the remaining ports. The event
that this port receives state $0$ and counter $b$ has probability
$\mu(0)\nu_\sigma(b)>0$. Conditioning on this event inserts the first marker and then follows
the designated path in \cref{def:composite-kernel}. The conditional cascade law is therefore the
marked pair law.
\end{proof}

\begin{lemma}[Support equality]
\label{thm:support-equality}
\lean{GraphMarkovMatching.Composite.exists_sim_of_muM_ne_zero}
\leanok
\uses{def:composite-kernel}
After forgetting the counters, the supports of the corresponding components on the two sides
agree, componentwise and at every height.
\end{lemma}

\begin{proof}
\leanok
We proceed by induction on the height and compare all corresponding component laws
simultaneously. At height zero, forgetting the counter leaves either the common state law $\mu$
or the point mass at $0$ on both sides. Assume the support identity through height $h$. The two
composite kernels may assign different counter names, but corresponding letters use the same
state law and the same straight or crossed child arrangement. The induction hypothesis identifies
the state supports of all child components. Applying the branch constructor therefore identifies
the state support at height $h+1$.
\end{proof}

\begin{corollary}[Exact pruning across the two laws]
\label{thm:exact-pruning}
\lean{GraphMarkovMatching.Composite.dead_screen_eq_zero}
\leanok
A screen whose cell is a component on one side and whose zero list holds the matching components
on the other side vanishes.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:support-equality, thm:pruning}
By \cref{thm:support-equality}, every point of positive source mass has a target configuration
with the same state at every vertex after counters are forgotten. Reflexivity makes the two state
labellings compatible under the identity pairing. Thus every supported source point is alive
against the matching target component, and the dead-event indicator vanishes. Integrating it gives
zero, as in \cref{thm:pruning}.
\end{proof}

\begin{definition}[The tagged alphabet and letter successors]
\label{def:composite-grammar}
\lean{GraphMarkovMatching.Composite.Letter}
\leanok
\uses{def:grammar, def:composite-kernel}
The tagged alphabet adds to the ordinary letters the marker letters carrying a declared pair and
a stage, with successors following the designated path.
\end{definition}

\begin{definition}[The live index of the tagged grammar]
\label{def:composite-index}
\lean{GraphMarkovMatching.Composite.screenUniv}
\leanok
\uses{def:composite-grammar, def:scracc}
The seeds are the fresh screens in either orientation, and the live index is the live part of
their closure under full successors, restriction of the zero list, and the two side maps.
\end{definition}

\begin{lemma}[Universe bound]
\label{thm:composite-universe}
\lean{GraphMarkovMatching.Composite.cLetterBox_card}
\leanok
\uses{def:composite-index}
With $\cL_N$ the formal envelope of one fresh name, the counter names $0,\dots,N$ and all marker
names with coordinates and stage in $\set{0,\dots,N}$,
$\abs{\cL_N}=1+(N+1)+(N+1)^3=n_{\cA}$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:universes}
The envelope contains one fresh name, $N+1$ counter names, and one marker name for each triple of
coordinates and stages in $\set{0,\dots,N}$. These three families are disjoint, so their
cardinalities add to $n_{\cA}=1+(N+1)+(N+1)^3$. A formal screen consists of one cell name, a
subset of the alphabet as its zero list, and either one normalisation name or no normalisation.
There are therefore $2^{n_{\cA}}$ choices of zero list and $n_{\cA}+1$ choices of
normalisation. This gives the required finite bound for the tagged screen universe.
\end{proof}

\begin{lemma}[Chart returns]
\label{thm:chart-returns}
\lean{GraphMarkovMatching.Composite.chartPorts_cases}
\leanok
\uses{def:renewal, def:composite-kernel}
The designated port depth satisfies $d(k)\in\mathrm{toF}(k)$, and the port depths of the chart of
a declared pair lie in the corresponding union of return-length sets.
\end{lemma}

\begin{proof}
\leanok
\uses{def:renewal}
We induct on $k$. For $k\leq3$, the designated port is reached in one step and
$1\in\mathrm{toF}(k)$. For $k\geq4$, the designated path enters the
$\floor{k/2}$ child cascade. The induction hypothesis gives its remaining depth, and the recursive
definition of $\mathrm{toF}(k)$ adds the initial step. Hence $d(k)\in\mathrm{toF}(k)$. A chart
for a declared pair is obtained by concatenating this designated path with one of its exposed
fresh-port paths. The same recursion places every resulting total depth in the union of the
corresponding return-length sets.
\end{proof}

\begin{lemma}[Realisation]
\label{thm:composite-realise}
\lean{GraphMarkovMatching.Composite.descend_F_iff_mem_sigmaS}
\leanok
On either side the fresh-to-fresh descent lengths are $\Sigma_S$, the descent lengths to $\Ff$
do not depend on the side, and every chain of $2N+2$ successor steps passes through $\Ff$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:chart-returns, thm:descend-realise}
For a common-core counter, the successors of $\Ff$ are the ordinary cascade pair on both sides.
Thus \cref{thm:descend-realise} realises all generators and identifies their concatenations with
$\Sigma_S$. For an exceptional entrance, the marker chain follows a chart. By
\cref{thm:chart-returns}, every port depth in that chart belongs to the same return monoid, so
inserting the marker chain does not create a side-dependent completion length. Consequently the
fresh-to-fresh and letter-to-fresh descent sets agree on both sides. Finally, every ordinary
cascade decreases its counter and every marker advances its finite stage, so within $2N+2$ steps
the path reaches $\Ff$.
\end{proof}

\begin{definition}[Anchored tagged letters and screens]
\label{def:composite-anchored}
\lean{GraphMarkovMatching.Composite.Anchored, GraphMarkovMatching.Composite.ScreenAnchored}
\leanok
\uses{def:anchored, thm:composite-realise}
A tagged letter is anchored at $\delta$ when $\delta+m\in\Sigma_S$ for every descent length $m$
to $\Ff$ on either side, and a tagged screen when its cell, zero list and normalisation are.
\end{definition}

\begin{lemma}[Anchoring propagates]
\label{thm:composite-anchor}
\lean{GraphMarkovMatching.Composite.Anchored.step}
\leanok
\uses{def:composite-anchored}
$\Ff$ is anchored at $\delta$ if and only if $\delta\in\Sigma_S$, and a successor of a letter
anchored at $\delta$ is anchored at $\delta+1$.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:anchor-step, thm:composite-realise}
The empty descent $\Ff\Downarrow_0\Ff$ shows that anchoring of $\Ff$ at $\delta$ implies
$\delta\in\Sigma_S$. Conversely, if $\delta\in\Sigma_S$, every fresh return has length in the
same monoid by \cref{thm:composite-realise}, and closure under addition gives anchoring. For the
successor clause, prepend one side-indexed successor step to any fresh completion of the successor.
This turns a completion of length $m$ at depth $\delta+1$ into one of length $m+1$ at depth
$\delta$. Anchoring of the original letter gives the required monoid membership.
\end{proof}

\begin{theorem}[Acyclicity of the screen steps]
\label{thm:composite-acyclic}
\lean{GraphMarkovMatching.Composite.no_live_step_cycleS}
\leanok
No cycle of full-successor edges passes through live anchored tagged screens, whether an edge
advances the cell by a common counter or through an exceptional entrance. The directed graph on
the live index is acyclic and the block matrix is nilpotent.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:composite-anchor, thm:nilpotence, thm:composite-realise,
thm:composite-universe}
Suppose that a directed cycle of full-successor edges passes through a live anchored tagged screen.
Repeat the cycle indefinitely. By \cref{thm:composite-anchor}, anchoring propagates at each step,
so this produces an infinite path of live anchored screens. The proof of
\cref{thm:nilpotence} applies because \cref{thm:composite-realise} gives the same renewal
alignment for the tagged grammar. This is a contradiction. The tagged live index is finite by
\cref{thm:composite-universe}, so its successor graph is acyclic. A topological ordering makes
the associated block matrix strictly triangular, hence nilpotent.
\end{proof}

\begin{lemma}[The weighted mixture tilt]
\label{thm:mixture-tilt-composite}
\lean{GraphMarkovMatching.Composite.invTilt_simDeg_cXiBar_le}
\leanok
\uses{def:composite-kernel}
At every pair point the restricted weight of the mixed pair law on one side is bounded by the
weighted sum over the common core and the exceptional entrances.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:mixture-tilt}
Fix a pair point. Its good degree is linear in the target law, and the target fresh law is the
mixture of the common-core components and the exceptional entrances with the original counter-law
coefficients. Hence the mixed good degree is the corresponding weighted sum of component good
degrees. If this sum vanishes, the restricted weight is zero. Otherwise divide by the positive
sum, as in \cref{thm:mixture-tilt}. The resulting component tilts retain the same mixture
coefficients, which yields the desired pointwise bound and hence the integrated inequality.
\end{proof}

\begin{lemma}[Retention of the source coefficients]
\label{thm:retention}
Every transfer-row expression is a sum of integrals against its source law, so expanding the
fresh source cell splits the row into component rows with the exact coefficients of the counter
law, on both the common core and the exceptional entrances.
\end{lemma}

\begin{proof}
\uses{thm:mixture-tilt-composite}
Expand the fresh source law inside each row integral before applying an inequality. Linearity of
the integral writes the row as the sum of its component rows, with exactly the common-core and
exceptional coefficients of the source counter law. For each component,
\cref{thm:mixture-tilt-composite} bounds the restricted target weight while retaining its target
coefficient. Thus neither the source nor the target mixture weights are lost when the transfer-row
estimate is applied.
\end{proof}

\begin{proposition}[The composite ledger]
\label{thm:composite-ledger}
\lean{GraphMarkovMatching.Composite.cPsi_step}
\leanok
\uses{thm:retention, thm:composite-acyclic, thm:psi-step, thm:e-step}
For two counter laws with a common core, the running coordinates formed from the two orientations
of the composite comparison satisfy the ordinary and screen step inequalities of
\cref{thm:psi-step} and \cref{thm:e-step} with the tagged step functions.
\end{proposition}

\begin{proof}
\leanok
\uses{thm:retention, thm:psi-step, thm:e-step, thm:exact-pruning}
Fix one orientation of the two-law comparison. By \cref{thm:retention}, every ordinary or screen
row decomposes into common-core rows and exceptional-entrance rows with the exact source and target
coefficients. Common-core rows satisfy the one-law estimates
\cref{thm:psi-step} and \cref{thm:e-step}. Exceptional rows follow the same estimates along their
finite marker charts, and \cref{thm:exact-pruning} removes a component whenever its matching
component appears in the zero list. Summing the component inequalities with their retained
coefficients gives the tagged ordinary and screen step functions. Repeating the calculation in
the opposite orientation gives the full composite ledger.
\end{proof}

\section{Proof of the two-law theorem}

\begin{proof}
\leanok
\proves{thm:composite-matching}
\uses{thm:composite-ledger, thm:composite-acyclic, thm:engine, thm:exact-pruning,
thm:composite-universe, thm:infinite}
Use the ordinary scalar and screen vector for both orientations of the composite comparison. The
ledger in \cref{thm:composite-ledger} supplies their ordinary and screen step inequalities. By
\cref{thm:composite-acyclic}, the tagged block matrix is nilpotent. The exact-pruning result
\cref{thm:exact-pruning} removes the diagonal component screens, while
\cref{thm:composite-universe} bounds the remaining finite index and therefore all row-sum and
invariant constants. We may now apply the abstract closure theorem \cref{thm:engine}. It gives the
uniform finite-height failure bound with the explicit composite constants. Marginal consistency
and the same compactness argument as in \cref{thm:infinite} pass the estimate to a single matching
automorphism on the infinite tree.
\end{proof}
